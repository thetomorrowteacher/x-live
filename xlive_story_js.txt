
    /* PATCH X-LIVE v0.39 -- Story Mode in the live room ------------------
       The platform's Story Mode writes each player's campaign progress to
       app_data key pflx_story_<playerId>. This panel reads those rows for
       whoever is in the room, so a host can see where the cohort actually
       is, and push a story beat to every joined player mid-session using
       the same xlStampPush path the slide pusher uses. */
    const XL_STORY = {"total":28,"acts":[{"id":"act0","n":"Act Zero","t":"Arrival","q":[["a0-brief","The Nexus Opens"],["a0-studio","Pick Your Studio"]]},{"id":"act1","n":"Act One","t":"The Alter Ego","q":[["a1-traits","Personality Trait Inventory"],["a1-forge","Character Forge"],["a1-profile","Character Profile"],["a1-board","Brand Board"],["a1-mint","Mint Your Alter Ego"]]},{"id":"act2","n":"Act Two","t":"BrandBuilder · Phase One","q":[["a2-brief","Client Briefing"],["a2-pick","Choose Your Client"],["a2-interview","The Client Interview"],["a2-cprofile","Client Profile"],["a2-empathy","Client Empathy Map"],["a2-cp3","Checkpoint 3 Submission"]]},{"id":"act3","n":"Act Three","t":"BrandBuilder · Phase Two","q":[["a3-r1","SparkLab Round One"],["a3-r2","SparkLab Round Two"],["a3-r3","SparkLab Round Three"],["a3-merge","The Combined Concept"],["a3-cp4","Checkpoint 4 Submission"]]},{"id":"act4","n":"Act Four","t":"The Developer's Workshop","q":[["a4-start","Starting State"],["a4-log","Daily Growth Log"],["a4-build","Build the Prototype"]]},{"id":"act5","n":"Act Five","t":"FeedForward Force","q":[["a5-course","FeedForward Mini Course"],["a5-give","Give Feedforward"],["a5-take","Take Feedforward"]]},{"id":"act6","n":"Act Six","t":"Exhibit","q":[["a6-design","Exhibit Design"],["a6-pitch","The Pitch"],["a6-show","Showcase"],["a6-eval","Self Evaluation and Future Steps"]]}]};
    const XL_STORY_QUEST = (function () {
      const m = {};
      XL_STORY.acts.forEach(function (a, ai) {
        a.q.forEach(function (q, qi) { m[q[0]] = { act: a, actIndex: ai, title: q[1], n: qi + 1 }; });
      });
      return m;
    })();

    // Pure: how far through the campaign a saved state is.
    function xlStoryPct(st) {
      if (!st || !st.done) return 0;
      const n = Object.keys(st.done).length;
      return Math.max(0, Math.min(100, Math.round(100 * n / XL_STORY.total)));
    }
    // Pure: the furthest act a player has finished a quest in, and what is next.
    function xlStoryStage(st) {
      const done = (st && st.done) || {};
      let far = -1;
      Object.keys(done).forEach(function (id) {
        const q = XL_STORY_QUEST[id];
        if (q && q.actIndex > far) far = q.actIndex;
      });
      let next = null;
      for (let i = 0; i < XL_STORY.acts.length && !next; i++) {
        const a = XL_STORY.acts[i];
        for (let j = 0; j < a.q.length; j++) {
          if (!done[a.q[j][0]]) { next = { act: a, title: a.q[j][1], id: a.q[j][0] }; break; }
        }
      }
      return {
        act: far >= 0 ? XL_STORY.acts[far] : null,
        next: next,
        done: Object.keys(done).length,
        pct: xlStoryPct(st),
        complete: !next
      };
    }
    // Pure: sort furthest-first, then by X-Coin earned, then by name.
    function xlStorySort(rows) {
      return rows.slice().sort(function (a, b) {
        if (b.stage.done !== a.stage.done) return b.stage.done - a.stage.done;
        if ((b.st.xc || 0) !== (a.st.xc || 0)) return (b.st.xc || 0) - (a.st.xc || 0);
        return String(a.name || '').localeCompare(String(b.name || ''));
      });
    }
    window.xlStoryPct = xlStoryPct; window.xlStoryStage = xlStoryStage; window.xlStorySort = xlStorySort;

    L.story = { byId: {}, at: 0, loading: false };

    // One batched PostgREST read for every id in the room. kvLoad is one key
    // per call, so this goes direct rather than N round trips.
    async function xlStoryLoad(ids) {
      if (!ids || !ids.length) { L.story.at = Date.now(); return; }
      L.story.loading = true;
      try {
        const keys = ids.map(function (id) { return encodeURIComponent('pflx_story_' + id); }).join(',');
        const r = await fetch(SUPABASE_URL + '/rest/v1/app_data?key=in.(' + keys + ')&select=key,data',
                              { headers: SB_HEADERS });
        if (r.ok) {
          const rows = await r.json();
          rows.forEach(function (row) {
            const id = String(row.key || '').replace(/^pflx_story_/, '');
            if (id && row.data) L.story.byId[id] = row.data;
          });
        }
      } catch (e) {}
      L.story.loading = false; L.story.at = Date.now();
      render();
    }
    window.xlStoryLoad = xlStoryLoad;

    // Who the panel is about: the joined room if a session is running,
    // otherwise everyone this host can see.
    function xlStoryAudience() {
      const s = (typeof xlRunningSession === 'function') ? xlRunningSession() : null;
      if (s && (s.liveParticipants || []).length) {
        return { label: 'In the room', live: true,
                 ids: s.liveParticipants.map(function (p) { return p.id; }) };
      }
      return { label: 'Your roster', live: false,
               ids: (L.roster || []).map(function (r) { return r.id; }).slice(0, 60) };
    }

    function xlStoryRows() {
      const aud = xlStoryAudience();
      const rows = aud.ids.map(function (id) {
        const p = (L.roster || []).find(function (r) { return r.id === id; }) || { id: id };
        const st = L.story.byId[id] || {};
        return { id: id, name: p.brand || p.name || id, image: p.image || null,
                 st: st, stage: xlStoryStage(st) };
      });
      return { aud: aud, rows: xlStorySort(rows) };
    }

    function rStory() {
      const got = xlStoryRows(), rows = got.rows, aud = got.aud;
      const started = rows.filter(function (r) { return r.stage.done > 0; });
      const avg = started.length
        ? Math.round(started.reduce(function (a, r) { return a + r.stage.pct; }, 0) / started.length) : 0;
      const s = (typeof xlRunningSession === 'function') ? xlRunningSession() : null;

      let h = '<div class="card"><div class="cardT">STORY MODE &mdash; ' + esc(aud.label) + '</div>' +
        '<div class="xl-hint" style="text-align:left;margin:0 0 10px">Live campaign progress from the platform. ' +
        started.length + ' of ' + rows.length + ' started &middot; average ' + avg + '% &middot; ' +
        (L.story.at ? 'read ' + Math.max(0, Math.round((Date.now() - L.story.at) / 1000)) + 's ago' : 'not read yet') +
        '</div>' +
        '<button class="bigbtn xl-sm" onclick="xlStoryRefresh()">' +
        (L.story.loading ? 'READING…' : 'REFRESH') + '</button></div>';

      h += '<div class="card"><div class="cardT">THE ROOM</div>';
      if (!rows.length) {
        h += '<div class="xl-hint">Nobody to show yet. Start a session, or wait for the roster to load.</div>';
      } else {
        h += rows.map(function (r) {
          const g = r.stage;
          const where = g.complete ? 'Season complete'
            : (g.act ? g.act.n + ' &middot; ' + esc(g.act.t) : 'Not started');
          const nx = g.next ? esc(g.next.title) : '&mdash;';
          return '<div class="xl-story-row">' +
            '<div class="xl-story-who">' +
              (r.image ? '<img src="' + esc(r.image) + '" alt="">' : '<span class="xl-story-dot"></span>') +
              '<div><b>' + esc(r.name) + '</b>' +
              '<i>' + where + (r.st.client ? ' &middot; client ' + esc(r.st.client) : '') + '</i></div>' +
            '</div>' +
            '<div class="xl-story-bar"><i style="width:' + g.pct + '%"></i></div>' +
            '<div class="xl-story-num">' + g.pct + '%</div>' +
            '<div class="xl-story-next">next: ' + nx + '</div>' +
            '<div class="xl-story-xc">' + (r.st.xc || 0) + ' XC</div>' +
          '</div>';
        }).join('');
      }
      h += '</div>';

      h += '<div class="card"><div class="cardT">PUSH A STORY BEAT</div>' +
        '<div class="xl-hint" style="text-align:left;margin:0 0 10px">Sends every joined player to this quest. ' +
        'They get the pull-in on the platform, same as a slide push.' +
        (s && s.status === 'active' ? '' : ' <b style="color:#ff8ba0">No session is running.</b>') + '</div>' +
        '<div class="xl-story-beats">' +
        XL_STORY.acts.map(function (a) {
          return '<div class="xl-story-act"><div class="xl-bs-k">' + esc(a.n) + ' &middot; ' + esc(a.t) + '</div>' +
            a.q.map(function (q) {
              return '<button class="bigbtn xl-sm ghost" onclick="xlStoryBeat(\'' + q[0] + '\')">' +
                esc(q[1]) + '</button>';
            }).join('') + '</div>';
        }).join('') +
        '</div></div>';
      return h;
    }
    window.rStory = rStory;

    function xlStoryRefresh() {
      const aud = xlStoryAudience();
      xlStoryLoad(aud.ids);
      render();
    }
    window.xlStoryRefresh = xlStoryRefresh;

    async function xlStoryBeat(questId) {
      const q = XL_STORY_QUEST[questId];
      if (!q) return;
      const s = (typeof xlRunningSession === 'function') ? xlRunningSession() : null;
      if (!s || s.status !== 'active') { toast('Start a session first.'); return; }
      xlStampPush(s, 'Story Mode: ' + q.title);
      if (inPlatform) xlPostParent({ type: 'pflx_story_beat', sessionId: s.id, questId: questId,
                                     act: q.act.id, title: q.title });
      if (typeof xlSfx === 'function') xlSfx('golive', 0.6);
      toast('📖 Pushed: ' + q.title);
      render();
      await saveSession(s);
    }
    window.xlStoryBeat = xlStoryBeat;

    // The six beats a host reaches for mid-show, on the SHOW CONTROL pad.
    function xlStoryPadHtml(s) {
      const quick = ['a0-brief', 'a1-traits', 'a1-board', 'a2-interview', 'a2-empathy', 'a3-r1'];
      return '<div class="xl-hint" style="margin:10px 0 6px">STORY BEAT &mdash; sends joined players to a quest</div>' +
        '<div class="xl-story-pad">' + quick.map(function (id) {
          const q = XL_STORY_QUEST[id];
          if (!q) return '';
          return '<button class="bigbtn xl-sm" onclick="xlStoryBeat(\'' + id + '\')">' + esc(q.title) + '</button>';
        }).join('') + '</div>';
    }
    window.xlStoryPadHtml = xlStoryPadHtml;
